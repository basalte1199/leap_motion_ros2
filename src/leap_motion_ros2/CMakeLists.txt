cmake_minimum_required(VERSION 3.16)
project(leap_motion_ros2)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Bone.msg" "msg/Finger.msg" "msg/Hand.msg" "msg/TrackingFrame.msg" DEPENDENCIES std_msgs geometry_msgs)
add_executable(leap_motion_node src/leap_motion_node.cpp)
target_compile_features(leap_motion_node PRIVATE cxx_std_17)
target_compile_options(leap_motion_node PRIVATE -Wall -Wextra)
find_path(LEAPC_INCLUDE_DIR LeapC.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(NOT LEAPC_INCLUDE_DIR)
  message(FATAL_ERROR "LeapC.h not found. Install the Ultraleap SDK first (scripts/install_service.sh).")
endif()
target_include_directories(leap_motion_node PRIVATE "${LEAPC_INCLUDE_DIR}")
ament_target_dependencies(leap_motion_node rclcpp std_msgs geometry_msgs visualization_msgs diagnostic_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(leap_motion_node "${typesupport}" ${CMAKE_DL_LIBS})
install(TARGETS leap_motion_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE THIRD_PARTY.md DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
